export const TWO_FACTOR_PENDING_KEY = 'two_factor_pending'
export type TwoFactorPending = { userId: number; expiresAt: number; attempts: number }

export function readPending(raw: unknown, now: number): PendingReadResult {
  if (typeof raw !== 'object' || raw === null) {
    return { status: 'none' }

## Shared Patterns

### Two-layer authorization
**Source:** `start/routes.ts` l.212-218 + `app_settings_controller.ts` l.61
**Apply to:** `GET /admin` (role admin + `PlatformSettingsPolicy.manage`), `POST /edit-mode` (role admin|manager|teacher + `CoursePolicy.edit` via `denies`).

### Session key helper
**Source:** `app/helpers/two_factor_pending.ts` (exported key constant + typed `{ userId }` state); consumed by `auth_controller.ts` via relative `../helpers/*.js` import.
**Apply to:** `edit_mode.ts`, `config/inertia.ts`, `edit_mode_controller.ts`, `auth_controller.ts`.

### Flash + named-route redirect
**Source:** `courses_controller.ts` l.212-227 (`session.flash` … `response.redirect().toRoute('courses.show', { id })`), `course_contents_controller.ts` l.122-124 (`denies` branch).
**Apply to:** `edit_mode_controller.ts`. French messages; flash surfaces through `FlashMessages` (`flash-toaster.tsx` l.16-28) with no extra wiring.

### Header reads props via `usePage()`
